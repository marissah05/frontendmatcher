### 1. CREATE ACTIVE SUMMARY LOGIC

Inside dashboard.tsx:

Create a useMemo:

const activeSummary = useMemo(() => {
  const counts: Record<string, number> = {};

  actives.forEach(a => {
    counts[a.id] = 0;
  });

  rounds.forEach(round => {
    round.pnms.forEach(pnm => {
      if (pnm.matchedWith) counts[pnm.matchedWith]++;
      if (pnm.secondMatch) counts[pnm.secondMatch]++;
    });
  });

  return actives.map(a => ({
    id: a.id,
    name: a.name,
    count: counts[a.id] || 0
  }));
}, [rounds, actives]);
